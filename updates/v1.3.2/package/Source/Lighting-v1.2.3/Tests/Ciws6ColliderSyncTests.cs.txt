 public class Component:Object {public GameObject gameObject;public Transform transform=>gameObject.transform;}
 public class Transform:Component {
  public Transform parent;public List<Transform> kids=new List<Transform>();public Vector3 localPosition {get;set;}public Vector3 localScale=new Vector3(1,1,1);public Quaternion rotation=Quaternion.Identity;public int writes;
  public new string name=>gameObject.name;public int childCount=>kids.Count;public Transform GetChild(int i)=>kids[i];
  public Matrix4x4 World=>Matrix4x4.CreateScale(localScale)*Matrix4x4.CreateFromQuaternion(rotation)*Matrix4x4.CreateTranslation(localPosition)*(parent==null?Matrix4x4.Identity:parent.World);
  public Vector3 TransformPoint(Vector3 v)=>System.Numerics.Vector3.Transform(v,World);
  public Vector3 InverseTransformPoint(Vector3 v){Matrix4x4.Invert(World,out var inverse);return System.Numerics.Vector3.Transform(v,inverse);}
  public Vector3 position=>TransformPoint(new Vector3());
  public void Add(Transform child){child.parent=this;kids.Add(child);}
 }

  public GameObject(string n){name=n;transform=new Transform{gameObject=this};}
  public Component[] GetComponents(Type t)=>components.FindAll(c=>t.IsAssignableFrom(c.GetType())).ToArray();
  public T Add<T>() where T:Component,new(){var c=new T{gameObject=this};components.Add(c);return c;}
 }

 static void Match(UnityEngine.Transform tr,UnityEngine.Transform model,double[] origin,double[] scale){
  var world=model.TransformPoint(V(origin[0],origin[1],origin[2]));tr.localPosition=tr.parent.InverseTransformPoint(world);
  Matrix4x4.Invert(tr.parent.World,out var inv);var m=model.World*inv;Matrix4x4.Decompose(m,out var s,out var r,out var t);tr.rotation=r;tr.localScale=V(s.X*scale[0],s.Y*scale[1],s.Z*scale[2]);
 }
